"""Reproduce local preparation with existing tools; no installation or network."""
import hashlib
import json
import os
from pathlib import Path
import platform
import shutil
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]
OUT = ROOT / 'output'
STAMP = '2026-10-30'
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
def write(name, data):
    (OUT / name).write_text(json.dumps(data, indent=2, sort_keys=True) + '\n')
def snapshot():
    return {p.relative_to(ROOT / 'source').as_posix(): {'sha256': sha(p), 'mode': oct(p.stat().st_mode & 0o777), 'symlink': p.is_symlink()} for p in sorted((ROOT / 'source').rglob('*')) if p.is_file()}
record = json.loads((ROOT / 'source-record.json').read_text())
before = snapshot()
write('evidence/source-before.json', before)
assert {p.removeprefix('editorial-writing/'): v['sha256'] for p, v in before.items() if p.startswith('editorial-writing/')} == record['files']
assert before['existing-plugin.json']['sha256'] == json.loads((ROOT / 'input-inventory.json').read_text())['source/existing-plugin.json']
results = []
env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1')
def run(label, args, cwd=ROOT, expected=0):
    result = subprocess.run(args, cwd=cwd, env=env, text=True, capture_output=True)
    entry = {'label': label, 'command': args, 'cwd': str(cwd.relative_to(ROOT)), 'exit_code': result.returncode, 'expected_exit_code': expected, 'status': 'pass' if result.returncode == expected else 'fail', 'stdout': result.stdout, 'stderr': result.stderr, 'observation_date': STAMP}
    results.append(entry)
    write('evidence/commands.json', results)
    return entry
py = sys.executable
run('authoring validator', [py, '-B', 'existing-authoring/scripts/quick_validate.py', 'source/editorial-writing'])
run('existing prose checker tests', [py, '-B', '-m', 'unittest', 'discover', '-s', 'source/editorial-writing/tests', '-v'])
run('existing exporter tests', [py, '-B', '-m', 'unittest', 'discover', '-s', '.agents/skills/skill-delivery/tests', '-v'])
files = ['SKILL.md', 'LICENSE', 'DEPENDENCIES.md', 'references/examples.md', 'references/audience/adult.md', 'references/audience/early-reader.md', 'references/medium/web.md', 'references/medium/longform.md', 'references/medium/game-ui.md', 'references/medium/game-dialogue.md', 'scripts/prose_lint.py', 'tests/test_prose_lint.py']
standalone = {p: 'source/editorial-writing/' + p for p in files}
plugin = {'plugin.json': 'source/existing-plugin.json', 'LICENSE': 'source/editorial-writing/LICENSE', 'DEPENDENCIES.md': 'source/editorial-writing/DEPENDENCIES.md'}
plugin.update({'skills/editorial-writing/' + p: s for p, s in standalone.items()})
helper = '.agents/skills/skill-delivery/scripts/export_skill.py'
for kind, name, mapping in [('skill', 'editorial-writing', standalone), ('plugin', 'editorial', plugin)]:
    config = {'version': 1, 'name': name, 'kind': kind, 'source_commit': record['base_revision'], 'files': mapping}
    config_path = 'config/' + kind + '.json'
    write(config_path, config)
    dest = OUT / ('standalone' if kind == 'skill' else 'plugin')
    if dest.exists():
        raise SystemExit('Refusing to overwrite existing candidate: ' + str(dest))
    run(kind + ' export', [py, '-B', helper, 'export', '--source-root', str(ROOT), '--config', str(OUT / config_path), '--output', str(dest)])
    run(kind + ' ZIP verification', [py, '-B', helper, 'verify', '--manifest', str(dest / 'manifest.json'), '--archive', str(dest / 'skill.zip')])
    # --installed-root is the helper's directory-check interface; this is staged, not installed.
    run(kind + ' staged directory verification', [py, '-B', helper, 'verify', '--manifest', str(dest / 'manifest.json'), '--installed-root', str(dest / 'payload')])
run('checker CLI on supplied synthetic examples', [py, '-B', 'source/editorial-writing/scripts/prose_lint.py', 'source/editorial-writing/references/examples.md', '--format', 'json'])
run('checker missing input', [py, '-B', 'source/editorial-writing/scripts/prose_lint.py', 'output/evaluation/absent.md'], expected=2)
import yaml
skill = (ROOT / 'source/editorial-writing/SKILL.md').read_text()
front = yaml.safe_load(skill.split('---', 2)[1])
assert front['name'] == 'editorial-writing' and 0 < len(front['description']) <= 1024
assert front['license'] == 'Apache-2.0'
manifest = json.loads((ROOT / 'source/existing-plugin.json').read_text())
assert manifest['name'] == 'editorial' and manifest['version'] == '0.1.1'
assert manifest['$schema'] == 'https://agent-plugins.org/schemas/1.0.0/plugin.schema.json'
assert manifest['license'] == 'Apache-2.0' and 'com.openai' in manifest['extensions']
write('evidence/structure.json', {'status': 'pass', 'scope': 'Basic fields inspected and asserted; not full plugin JSON Schema or Vally validation', 'frontmatter': front, 'skill_lines': len(skill.splitlines()), 'plugin_version': manifest['version'], 'extensions_preserved': list(manifest['extensions'])})
write('evidence/environment.json', {'date': STAMP, 'python': sys.version, 'platform': platform.platform(), 'PyYAML': yaml.__version__, 'available_executables': {n: shutil.which(n) for n in ['python3','node','skills','copilot','codex','gh']}, 'agent': 'Codex (host session); exact backend model/build not exposed; no independent runtime trial', 'installs': 'None', 'node_probe': 'node --version failed: missing libsimdjson.29.dylib; runtime not repaired', 'official_document_snapshots': record['official_sources'], 'authoring_validator_sha256': sha(ROOT / 'existing-authoring/scripts/quick_validate.py'), 'export_helper_sha256': sha(ROOT / helper)})
after = snapshot()
write('evidence/source-after.json', after)
assert before == after, 'Source inventory/bytes/modes changed'
write('evidence/source-integrity.json', {'status': 'pass', 'source_file_count': len(before), 'all_skill_files_match_supplied_source_record': True, 'plugin_matches_input_inventory': True, 'before_equals_after_inventory_hashes_modes': before == after, 'source_revision': record['base_revision'], 'revision_limit': 'Supplied snapshot; no .git checkout or remote byte verification. Commit is a provenance label, not proof of remote identity.', 'excluded_from_both_payloads': ['scripts/__pycache__/prose_lint.cpython-312.pyc', 'tests/__pycache__/test_prose_lint.cpython-312.pyc']})
write('evidence/artifact-hashes.json', {str(p.relative_to(OUT)): sha(p) for sub in ['standalone', 'plugin'] for p in sorted((OUT / sub).rglob('*')) if p.is_file()})
print(json.dumps({'commands': len(results), 'failed': [r['label'] for r in results if r['status'] != 'pass'], 'source_unchanged': before == after}, indent=2))
